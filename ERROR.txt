        // ─────────────────────────────────────────────
        // TARGET WORLD SIZE FOR FRAME (in Three.js units)
        // Everything else scales relative to this.
        const FRAME_TARGET = 400; // frame longest axis = 400 units (enlarged for visibility)

        // Once frame is loaded we fill these in:
        let frameBox = null;   // THREE.Box3 in world space (after scale applied)
        let frameScale = 1;    // scale factor applied to frame mesh

        // Snap definitions as FRACTIONS of frame bounding box.
        // fx/fy/fz = fraction along each axis (0=min, 0.5=center, 1=max)
        // These represent logical bicycle positions.
        // fx/fy/fz  : snap position as fraction of frame bounding box (0=min edge, 1=max edge)
        // orient    : 'wheel' = auto-detect thin axis; 'none' = use rx/ry as-is
        // targetFrac: part's longest axis = this fraction × frame's longest axis
        //
        // FRAME ORIENTATION (observed from screenshots):
        //   The fork/front end of the frame sits at the LOW-X side (frameBox.min.x)
        //   The rear triangle sits at the HIGH-X side (frameBox.max.x)
        const SLOT_DEF = {
            // 1. Frame: small anticlockwise tilt via rz
            frame: { fx: 0.5, fy: 0.5, fz: 0.5, orient: 'none', rx: 0, ry: 0, rz: -Math.PI / 14, targetFrac: 1.0 },
            // Wheels: auto-detect thin axis; 2. backWheel extraRy flipped 180° for correct orientation
            frontWheel: { fx: 0.05, fy: 0.10, fz: 0.5, orient: 'wheel', extraRy: Math.PI / 2, targetFrac: 0.44 },
            backWheel: { fx: 0.95, fy: 0.10, fz: 0.5, orient: 'wheel', extraRy: -Math.PI / 2, targetFrac: 0.44 },
            // 3. Seat: ry:PI flips it front-facing (about-turn horizontally)
            // Saddle: very light anticlockwise tilt from horizontal
            seat: { fx: 0.78, fy: 1.05, fz: 0.5, orient: 'none', rx: -Math.PI / 22, ry: Math.PI, targetFrac: 0.21 },
            fork: { fx: 0.10, fy: 0.58, fz: 0.5, orient: 'none', rx: 0, ry: 0, targetFrac: 0.34 },
            // Handlebar: scale increased
            handlebar: { fx: 0.10, fy: 0.88, fz: 0.5, orient: 'none', rx: 0, ry: 0, targetFrac: 0.36 },
            // Chain: vertical about-turn (ry:-PI/2)
            chain: { fx: 0.50, fy: 0.32, fz: 0.5, orient: 'none', rx: 0, ry: -Math.PI / 2, targetFrac: 0.48 },
            // 8. Pedal: scale decreased
            pedalCrank: { fx: 0.50, fy: 0.10, fz: 0.58, orient: 'none', rx: 0, ry: Math.PI / 2, targetFrac: 0.11 },
            // 7. Sprocket: scale increased
            pedalSprocket: { fx: 0.50, fy: 0.08, fz: 0.5, orient: 'none', rx: Math.PI / 2, ry: 0, targetFrac: 0.30 },
            // 6. Tire levers: scale increased
            backTireBolt: { fx: 0.95, fy: 0.10, fz: 0.5, orient: 'none', rx: 0, ry: 0, targetFrac: 0.14 },
            tireBolt: { fx: 0.05, fy: 0.10, fz: 0.5, orient: 'none', rx: 0, ry: 0, targetFrac: 0.14 },
            // Handlebar bushes: 1 spawned, forward lean to fit head tube
            tireBush: { fx: 0.10, fy: 0.88, fz: 0.5, orient: 'none', rx: -Math.PI / 10, ry: 0, rz: 0, targetFrac: 0.04 },
        };

        function init() {
            scene = new THREE.Scene();
            // Premium gradient-like dark background
            scene.background = new THREE.Color(0x0f1923);
            scene.fog = new THREE.FogExp2(0x0f1923, 0.0008);

            // Enhanced grid with dark theme
            const grid = new THREE.GridHelper(1600, 40, 0x1e3a5f, 0x0d2137);
            grid.position.y = -220;
            scene.add(grid);

            const W = window.innerWidth - 240, H = window.innerHeight;
            camera = new THREE.PerspectiveCamera(42, W / H, 0.1, 30000);
            camera.position.set(0, 150, 800);
            camera.lookAt(0, 0, 0);

            renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false });
            renderer.setSize(W, H);
            renderer.setPixelRatio(window.devicePixelRatio);
            // Premium tone mapping for photorealistic look
            renderer.toneMapping = THREE.ACESFilmicToneMapping;
            renderer.toneMappingExposure = 1.2;
            renderer.shadowMap.enabled = true;
            renderer.shadowMap.type = THREE.PCFSoftShadowMap;
            document.getElementById('viewport').appendChild(renderer.domElement);

            // Rich multi-light setup
            const ambient = new THREE.AmbientLight(0x334466, 0.6);
            scene.add(ambient);

            // Key light (warm sun)
            const sun = new THREE.DirectionalLight(0xfff4e0, 1.6);
            sun.position.set(300, 600, 400);
            sun.castShadow = true;
            sun.shadow.mapSize.width = 2048;
            sun.shadow.mapSize.height = 2048;
            sun.shadow.camera.near = 0.5;
            sun.shadow.camera.far = 3000;

        function drag(ev, file, slot, color) {

        // ── Duplicate-part toast helper ───────────────
        let _dupToastTimer = null;
        function showDuplicateToast(partName) {
            const toast = document.getElementById('dup-toast');
            document.getElementById('dup-toast-msg').textContent =
                partName + ' is already added! Delete it first to replace.';
            toast.style.display = 'block';
            clearTimeout(_dupToastTimer);
            _dupToastTimer = setTimeout(() => { toast.style.display = 'none'; }, 3000);
        }

        // Slots where duplicates are always blocked (single-instance parts)
        const SINGLE_SLOTS = new Set([
            'frame', 'frontWheel', 'backWheel', 'seat', 'fork',
            'handlebar', 'chain', 'pedalCrank', 'pedalSprocket', 'backTireBolt', 'tireBush'
        ]);

        // Friendly display names for the toast
        const PART_DISPLAY_NAME = {
            frame: 'Frame', frontWheel: 'Front Wheel', backWheel: 'Back Wheel',
            seat: 'Saddle', fork: 'Fork', handlebar: 'Handlebar', chain: 'Chain',
            pedalCrank: 'Pedals', pedalSprocket: 'Sprocket',
            backTireBolt: 'Tire Levers', tireBush: 'Handlebar Bushes'
        };

        // ── Drop on viewport ─────────────────────────
        function onDrop(ev) {
            if (isLocked) return;
            ev.preventDefault();
            const raw = ev.dataTransfer.getData("text/plain");
            if (!raw) return;
            const { file, slot, color } = JSON.parse(raw);

            // ── Duplicate guard ──────────────────────────────────────────────
            if (SINGLE_SLOTS.has(slot)) {
                const alreadyAdded = allParts.some(m =>
                    m.userData.slot === slot ||
                    // pedal pair uses same slot for both meshes
                    (slot === 'pedalCrank' && m.userData.slot === 'pedalCrank') ||
                    // tire lever pair uses backTireBolt / tireBolt
                    (slot === 'backTireBolt' && (m.userData.slot === 'backTireBolt' || m.userData.slot === 'tireBolt'))
                );
                if (alreadyAdded) {
                    showDuplicateToast(PART_DISPLAY_NAME[slot] || slot);
                    return;
                }
            }

            const url = uploadedFiles[file.toLowerCase()] || file;
            loadPart(url, slot, color);
        }

        // ── Load & auto-scale STL ────────────────────
        function loadPart(url, slot, color) {
            setStatus('Loading…');
            new THREE.STLLoader().load(url, function (geo) {
                geo.center();
                geo.computeVertexNormals();
                geo.computeBoundingBox();

                const size = new THREE.Vector3();
                geo.boundingBox.getSize(size);
                const longest = Math.max(size.x, size.y, size.z);

                const mat = new THREE.MeshStandardMaterial({ color, metalness: 0.45, roughness: 0.5, envMapIntensity: 0.8 });
                const mesh = new THREE.Mesh(geo, mat);
                mesh.castShadow = true;
                mesh.receiveShadow = true;

                if (slot === 'frame') {
                    // Scale frame to FRAME_TARGET units
                    frameScale = FRAME_TARGET / longest;
                    mesh.scale.setScalar(frameScale);
                    mesh.position.set(0, 0, 0);

                    // Compute world-space bounding box for snap reference
                    mesh.updateMatrixWorld(true);
                    frameBox = new THREE.Box3().setFromObject(mesh);

                    // Hide hint
                    const fh = document.getElementById('frame-hint');
                    if (fh) fh.style.display = 'none';
                    setStatus('Frame loaded ✓ — now drop other parts');
                } else {
                    if (!frameBox) {
                        setStatus('⚠️ Drop the Frame first!');
                        return;
                    }
                    const def = SLOT_DEF[slot] || SLOT_DEF.frame;
                    const fbSize = new THREE.Vector3();
                    frameBox.getSize(fbSize);
                    const refLen = Math.max(fbSize.x, fbSize.y, fbSize.z);

                    // Scale: part's longest axis = targetFrac × frame's longest axis
                    const targetSize = def.targetFrac * refLen;
                    mesh.scale.setScalar(targetSize / longest);

                    // Orientation
                    applyOrientation(mesh, geo, def, slot);

                    // Snap position inside frame bbox
                    const snapX = frameBox.min.x + def.fx * fbSize.x;
                    const snapY = frameBox.min.y + def.fy * fbSize.y;
                    const snapZ = frameBox.min.z + def.fz * fbSize.z;

                    // ── Pedal: spawn mirrored pair (×2) ─────────────────────
                    if (slot === 'pedalCrank') {
                        const zOff = fbSize.z * 0.12;
                        mesh.position.set(snapX, snapY, snapZ + zOff);
                        mesh.rotation.y = Math.PI / 2;
                        mesh.userData.slot = slot;
                        scene.add(mesh); allParts.push(mesh);

                        const meshL = new THREE.Mesh(geo, mat.clone());
                        meshL.scale.setScalar(targetSize / longest);
                        meshL.position.set(snapX, snapY, snapZ - zOff);
                        meshL.rotation.y = -Math.PI / 2;
                        meshL.userData.slot = slot;
                        scene.add(meshL); allParts.push(meshL);

                        setStatus('Pedal pair snapped ✓ — drag to fine-tune');
                        rebuildDrag(); return;
                    }

        // ── Smart orientation ────────────────────────
        // For wheels: finds the THINNEST dimension (= the axle direction) and
        // rotates the mesh so that axis aligns with world-X (left↔right axle).
        // This makes the wheel stand upright and face forward regardless of how
        // the STL was originally exported.
        function applyOrientation(mesh, geo, def, slot) {
            if (def.orient === 'wheel') {
                const s = new THREE.Vector3();
                geo.boundingBox.getSize(s);     // geometry-space dims before scale
                const minVal = Math.min(s.x, s.y, s.z);

                // Step 1 – stand the wheel upright by aligning its thin axis with Z
                // (wheel face will be in XY plane, visible from front/back)
                if (minVal === s.z) {
                    // Already thin along Z → no rotation needed for upright
                    mesh.rotation.set(0, 0, 0);
                } else if (minVal === s.y) {
                    // Thin along Y → rotate -90° around X so Y becomes Z
                    mesh.rotation.set(-Math.PI / 2, 0, 0);
                } else {
                    // Thin along X → rotate 90° around Y so X becomes Z
                    mesh.rotation.set(0, Math.PI / 2, 0);
                }

                // Step 2 – apply the extra Y rotation (extraRy) to turn the wheel
                // so it faces the correct direction (right-turn = PI/2)
                mesh.rotation.y += (def.extraRy || 0);

            } else {
                // Use explicit rx/ry/rz from SLOT_DEF
                mesh.rotation.set(def.rx || 0, def.ry || 0, def.rz || 0);
            }
        }

        function guessSlot(n) {
            n = n.toLowerCase();
            if (n.includes('frame')) return 'frame';
            if (n.includes('sadle') || n.includes('saddle') || n.includes('seat')) return 'seat';
            if (n.includes('fork')) return 'fork';
            if (n.includes('handle') || n.includes('bar')) return 'handlebar';
            if (n.includes('chain')) return 'chain';
            if (n.match(/^4\./) || (n.includes('rim') && n.includes('4'))) return 'frontWheel';
            if (n.match(/^5\./) || (n.includes('rim') && n.includes('5'))) return 'backWheel';
            if (n.includes('rim') || n.includes('tire') || n.includes('wheel')) return 'frontWheel';
            if (n.includes('liver') || n.includes('lever')) return 'backTireBolt';
            if (n.includes('bush')) return 'tireBush';
            if (n.match(/^8\./) || n.includes('pedal') || n.includes('crank')) return 'pedalCrank';
            if (n.match(/^6\./) || n.includes('spracket') || n.includes('sprocket')) return 'pedalSprocket';
            return 'frame';
        }

        // ── Drag controls (in-scene repositioning) ───
        function rebuildDrag() {
            if (dragControls) dragControls.dispose();
            dragControls = new THREE.DragControls(allParts, camera, renderer.domElement);
            dragControls.addEventListener('dragstart', () => { if (!isLocked) orbit.enabled = false; });
            dragControls.addEventListener('drag', e => { if (!isLocked) checkTrash(e.object); });
            dragControls.addEventListener('dragend', e => { orbit.enabled = true; maybeDelete(e.object); });
            dragControls.enabled = !isLocked;
        }

        // ── Trash detection ──────────────────────────
        function checkTrash(obj) {
            const v = obj.position.clone().project(camera);
            const sx = (v.x + 1) * (window.innerWidth - 240) / 2 + 240;
            document.getElementById('trash-zone').classList.toggle('hot', sx < 260);
        }
        function maybeDelete(obj) {
            const v = obj.position.clone().project(camera);
            const sx = (v.x + 1) * (window.innerWidth - 240) / 2 + 240;
            if (sx < 260) {
                if (obj.userData.slot === 'frame') { frameBox = null; frameScale = 1; }
                scene.remove(obj);
                allParts.splice(allParts.indexOf(obj), 1);
                rebuildDrag();
                setStatus('Part deleted');
            }
            document.getElementById('trash-zone').classList.remove('hot');
        }

        // ── Right-click rotate individual part ───────
        function onRightDrag(e) {
            if (e.buttons !== 2 || isLocked) return;
            const rect = renderer.domElement.getBoundingClientRect();
            mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
            mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
            raycaster.setFromCamera(mouse, camera);
            const hits = raycaster.intersectObjects(allParts);
            if (hits.length > 0) {
                orbit.enabled = false;
                hits[0].object.rotation.y += e.movementX * 0.012;
                hits[0].object.rotation.x += e.movementY * 0.012;
            }
        }
